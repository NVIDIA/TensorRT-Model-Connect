// @title: FLUX diffusion inference pipeline
// @description: Prompt conditioning and initial latent noise enter a denoiser and scheduler loop; after the configured steps, the final latents pass through the VAE decoder and become ImageResult pixels or frames.
digraph FluxDenoisingPipeline {
  graph [
    bgcolor="#fbfcfe",
    pad="0.30",
    nodesep="0.44",
    ranksep="0.54",
    rankdir="TB",
    splines="polyline",
    fontname="Arial",
    labelloc="t",
    label=<<B>Diffusion repeats over latent tensors</B><BR/><FONT POINT-SIZE="16">The loop ends after configured denoising steps, not an EOS token.</FONT>>,
    fontsize="27",
    fontcolor="#111827"
  ];
  node [
    shape="rect",
    style="rounded,filled",
    color="#64748b",
    penwidth="1.8",
    fillcolor="#ffffff",
    fontname="Arial",
    fontsize="17",
    fontcolor="#111827",
    margin="0.22,0.14"
  ];
  edge [color="#64748b", penwidth="2.2", arrowsize="0.8", fontname="Arial", fontsize="16", fontcolor="#334155"];

  prompt [fillcolor="#eff6ff", color="#2563eb", label=<<B>Prompt text</B>>];
  encoder [label=<<B>Text encoder</B>>];
  conditioning [label=<<B>Conditioning embeddings</B>>];
  noise [fillcolor="#eff6ff", color="#2563eb", label=<<B>Initial latent noise</B>>];
  { rank=same; prompt; noise; }
  prompt -> encoder -> conditioning;

  subgraph cluster_loop {
    label=<<B>Repeat for the configured denoising steps</B>>;
    color="#b45309";
    penwidth="2";
    style="rounded,filled";
    fillcolor="#fffbeb";
    fontname="Arial";
    fontsize="18";
    fontcolor="#78350f";
    denoiser [fillcolor="#ecfccb", color="#4d7c0f", label=<<B>Denoiser engine</B>>];
    scheduler [fillcolor="#ffffff", color="#b45309", label=<<B>Scheduler step</B>>];
    denoiser -> scheduler [label="update latents"];
    scheduler -> denoiser [label="next step", constraint=false];
  }

  conditioning -> denoiser;
  noise -> denoiser;
  latents [label=<<B>Final latents</B>>];
  vae [fillcolor="#ecfccb", color="#4d7c0f", label=<<B>VAE decoder</B>>];
  result [fillcolor="#f5f3ff", color="#7c3aed", label=<<B>ImageResult</B><BR/><FONT POINT-SIZE="16">pixels or frames</FONT>>];
  scheduler -> latents [label="last step"];
  latents -> vae -> result;
}
