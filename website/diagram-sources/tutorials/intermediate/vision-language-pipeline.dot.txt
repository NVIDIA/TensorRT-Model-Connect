// @title: Vision-language inference pipeline
// @description: Text tokens and image embeddings merge before the runtime follows the decoder layout recorded in the bundle, samples generated tokens, and returns TextResult.
digraph VisionLanguagePipeline {
  graph [
    bgcolor="#fbfcfe",
    pad="0.30",
    nodesep="0.42",
    ranksep="0.62",
    rankdir="TB",
    splines="polyline",
    fontname="Arial",
    labelloc="t",
    label=<<B>Text and image inputs meet at the decoder</B><BR/><FONT POINT-SIZE="16">Inspect the bundle instead of assuming one decoder layout for every Qwen-VL build.</FONT>>,
    fontsize="27",
    fontcolor="#111827"
  ];
  node [
    shape="rect",
    style="rounded,filled",
    color="#64748b",
    penwidth="1.8",
    fillcolor="#ffffff",
    fontname="Arial",
    fontsize="17",
    fontcolor="#111827",
    margin="0.20,0.14"
  ];
  edge [color="#64748b", penwidth="2.2", arrowsize="0.8", fontname="Arial", fontsize="16", fontcolor="#334155"];

  prompt [fillcolor="#eff6ff", color="#2563eb", label=<<B>Text prompt</B>>];
  tokenizer [label=<<B>Tokenizer</B><BR/><FONT POINT-SIZE="16">text token IDs</FONT>>];
  image [fillcolor="#eff6ff", color="#2563eb", label=<<B>Image pixels</B>>];
  preprocess [label=<<B>Image preprocessor</B><BR/><FONT POINT-SIZE="16">resize · normalize · layout</FONT>>];
  vision [fillcolor="#ecfccb", color="#4d7c0f", label=<<B>Vision engine</B>>];
  embeddings [label=<<B>Image embeddings</B>>];

  { rank=same; prompt; image; }
  { rank=same; tokenizer; preprocess; }
  { rank=same; vision; embeddings; }
  prompt -> tokenizer;
  image -> preprocess -> vision -> embeddings;

  merge [fillcolor="#f5f3ff", color="#7c3aed", label=<<FONT POINT-SIZE="19"><B>Merge prompt and image embeddings</B></FONT><BR/><FONT POINT-SIZE="16">at model-owned placeholder positions</FONT>>];
  tokenizer -> merge;
  embeddings -> merge;

  layout [shape="diamond", style="filled", fillcolor="#fffbeb", color="#b45309", label=<<B>Decoder layout<BR/>in this bundle</B>>];
  merge -> layout;

  combined [fillcolor="#ecfccb", color="#4d7c0f", label=<<B>Single or dual-profile</B><BR/><FONT POINT-SIZE="16">decoder engine</FONT>>];
  prefill [fillcolor="#ecfccb", color="#4d7c0f", label=<<B>Split prefill plan</B><BR/><FONT POINT-SIZE="16">prefill_engine_plan</FONT>>];
  decode [fillcolor="#ecfccb", color="#4d7c0f", label=<<B>Split decode plan</B><BR/><FONT POINT-SIZE="16">engine_plan</FONT>>];
  { rank=same; combined; prefill; }
  layout -> combined [label="fallback layout"];
  layout -> prefill [label="qualified split"];
  prefill -> decode;

  sampler [label=<<B>Sampler</B><BR/><FONT POINT-SIZE="16">next-token selection</FONT>>];
  output [fillcolor="#f5f3ff", color="#7c3aed", label=<<FONT POINT-SIZE="19"><B>TextResult</B></FONT>>];
  combined -> sampler;
  decode -> sampler;
  sampler -> output;
}
