// @title: Shared task pipeline pattern
// @description: Every task accepts typed input, performs task-owned preprocessing, executes one or more TensorRT engines, postprocesses the outputs, and returns a typed result; the repeated work differs by task.
digraph TaskPipelinePatterns {
  graph [
    bgcolor="#fbfcfe",
    pad="0.30",
    nodesep="0.34",
    ranksep="0.68",
    rankdir="TB",
    splines="ortho",
    fontname="Arial",
    labelloc="t",
    label=<<B>One public pattern, task-owned behavior</B><BR/><FONT POINT-SIZE="16">The bundle and plugin contract stay stable while each task owns its processing loop.</FONT>>,
    fontsize="27",
    fontcolor="#111827"
  ];
  node [
    shape="rect",
    style="rounded,filled",
    color="#64748b",
    penwidth="1.8",
    fillcolor="#ffffff",
    fontname="Arial",
    fontsize="17",
    fontcolor="#111827",
    margin="0.20,0.14"
  ];
  edge [color="#64748b", penwidth="2.2", arrowsize="0.8", fontname="Arial", fontsize="16", fontcolor="#334155"];

  input [fillcolor="#eff6ff", color="#2563eb", label=<<FONT POINT-SIZE="19"><B>Typed input</B></FONT><BR/><FONT POINT-SIZE="16">text · pixels · audio · numbers</FONT>>];
  preprocess [label=<<FONT POINT-SIZE="19"><B>Preprocess</B></FONT><BR/><FONT POINT-SIZE="16">task-owned input preparation</FONT>>];
  engines [fillcolor="#ecfccb", color="#4d7c0f", label=<<FONT POINT-SIZE="19"><B>TensorRT engines</B></FONT><BR/><FONT POINT-SIZE="16">one engine or a component pipeline</FONT>>];
  postprocess [label=<<FONT POINT-SIZE="19"><B>Postprocess</B></FONT><BR/><FONT POINT-SIZE="16">decode task-specific outputs</FONT>>];
  result [fillcolor="#f5f3ff", color="#7c3aed", label=<<FONT POINT-SIZE="19"><B>Typed result</B></FONT><BR/><FONT POINT-SIZE="16">text · image · audio · forecast</FONT>>];

  { rank=same; input; preprocess; engines; }
  { rank=same; postprocess; result; }
  input -> preprocess -> engines [constraint=false];
  engines -> postprocess;
  postprocess -> result [constraint=false];

  token [fillcolor="#ffffff", label=<<B>Token decode</B><BR/><FONT POINT-SIZE="16">repeat until stop condition</FONT>>];
  denoise [fillcolor="#ffffff", label=<<B>Denoising</B><BR/><FONT POINT-SIZE="16">repeat for configured steps</FONT>>];
  chunks [fillcolor="#ffffff", label=<<B>Audio chunks</B><BR/><FONT POINT-SIZE="16">update streaming state</FONT>>];
  single [fillcolor="#ffffff", label=<<B>Single pass</B><BR/><FONT POINT-SIZE="16">vision or forecasting</FONT>>];

  { rank=same; token; denoise; chunks; single; }
  engines -> token [style=dashed];
  engines -> denoise [style=dashed];
  engines -> chunks [style=dashed];
  engines -> single [style=dashed];
  token -> denoise -> chunks -> single [style=invis, weight=10];
  input -> token [style=invis, weight=20];
  preprocess -> denoise [style=invis, weight=20];
  postprocess -> chunks [style=invis, weight=20];
  result -> single [style=invis, weight=20];
}
